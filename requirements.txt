# ── MARK LV — Python dependencies ─────────────────────────────────────────────
# pip skips any line whose sys_platform marker does not match your OS, so this
# one file serves Windows, macOS and Linux. `python setup.py` runs it for you.

# Version bounds: the lower bound is what the app is developed against, the
# upper bound stops the next MAJOR release of a dependency from breaking every
# fresh clone on the day it ships. Minor and patch updates flow in freely.

# software with QPainter and posed with numpy, so the face costs no extra

# ── Video on the HUD ──────────────────────────────────────────────────────────
# Qt plays local files and direct media URLs on its own — QtMultimedia comes
# with PyQt6, so the player itself costs nothing. A YouTube page is not a video
# file, though, and yt-dlp is what turns one into a stream Qt can play. Without
# it the HUD still plays files and URLs, and YouTube links open in the browser
# with an explanation instead of failing.
#
# Deliberately UNBOUNDED, unlike everything else here. YouTube changes what it
# serves without notice and yt-dlp answers with a release; an upper bound would
# eventually pin a fresh clone to a version YouTube has already broken, which is
# the opposite of what bounds are for. Keep it current: pip install -U yt-dlp
yt-dlp

# ── Automation / input control ────────────────────────────────────────────────
pyautogui
pyperclip

# ── Vision & media ────────────────────────────────────────────────────────────
pillow>=10,<13
opencv-python>=4.8,<5
mss>=9,<11

# ── Screen targeting ──────────────────────────────────────────────────────────
# rapidocr-onnxruntime is the FALLBACK text detector for core/ui_grounding. On
# Windows the OS engine above is used instead — it is roughly 70x faster — so
# this only runs where winsdk is unavailable, or when the OCR language pack is
# missing. It is a large download (an ONNX runtime plus the PP-OCRv4 models), so
# it is imported lazily and a missing install disables exactly one fallback
# layer, never the app.
rapidocr-onnxruntime

# ── Plugin extras ─────────────────────────────────────────────────────────────
# Plugins are downloaded one file at a time from the community, so their
# dependencies are installed UP FRONT: dropping a .py into plugins/ should make
# it work on the next launch, not send you back to pip. Every one of these is
# imported inside a try/except at the point of use, so an uninstalled extra
# disables exactly one plugin and never touches the rest of the app.
google-api-python-client     # gmail_control, calendar_control — OAuth
google-auth-oauthlib         # gmail_control, calendar_control — OAuth
tinytuya                     # home_assistant — Tuya / Smart Life bulbs
paho-mqtt                    # printer_control — MQTT printers
openpyxl                     # excel_writer — spreadsheet output
youtube-transcript-api       # youtube_video — transcript fetch for summaries
pygetwindow                  # game_updater, chat_takeover — window geometry

# ── Windows-only (pip skips these automatically on macOS / Linux) ─────────────
comtypes;   sys_platform == "win32"
pycaw;      sys_platform == "win32"
win10toast; sys_platform == "win32"
pywin32;    sys_platform == "win32"
wmi;        sys_platform == "win32"
# pywinauto drives the Windows accessibility tree. Three plugins read it:
# screen_agent (to aim without a model call), the WhatsApp pair, and
# game_updater (Steam's install dialog).
pywinauto;  sys_platform == "win32"
# uiautomation is what core/ui_grounding actually aims with. It reads the same
# accessibility tree as pywinauto but is far cheaper per query, which matters
# because it runs inside the click path — a button resolves in ~10-40 ms with no
# model call, which is what makes an exact click feel instant. pywinauto stays
# as the fallback for the plugins above and for when this is not installed.
uiautomation;  sys_platform == "win32"
# winsdk exposes Windows.Media.Ocr, the OCR engine built into Windows. It reads
# a full 1080p window in ~0.1 s where the ONNX alternative below takes ~7 s, so
# it is the one used whenever it is available. No model download: it uses the
# OCR language pack the OS already has.
winsdk;  sys_platform == "win32"
